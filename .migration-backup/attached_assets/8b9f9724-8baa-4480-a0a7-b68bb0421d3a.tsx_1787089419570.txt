function useFonts() {

    const tag = document.createElement("style");
    tag.innerHTML = FONTS;

    return () => document.head.removeChild(tag);

}

function RequestFlow({ initialCat, onClose, onSubmit }) {
  const [step, setStep] = useState(0);
  const [cat, setCat] = useState(initialCat);
  const [site, setSite] = useState("Lara Konakları");
  const [blok, setBlok] = useState("B Blok");
  const [daire, setDaire] = useState("18");
  const [title, setTitle] = useState("");
  const [desc, setDesc] = useState("");
  const [priority, setPriority] = useState("acil");
  const [slot, setSlot] = useState(TIME_SLOTS[2]);
  const [photoAdded, setPhotoAdded] = useState(false);
  const [sent, setSent] = useState(false);

  const catObj = CATEGORIES.find((c) => c.key === cat);
  const canStep1 = !!cat;
  const canStep2 = title.trim().length > 2;

  const steps = ["Hizmet", "Detaylar", "Onay"];

  if (sent) {
    return <SearchingScreen catLabel={catObj?.label} onClose={onClose} onDone={() => onSubmit({ title, cat, catObj, priority, slot, site, blok, daire })} />;
  }

  return (
    <div style={s.screen}>
      <TopBar title="Talep Oluştur" onBack={onClose} />
      <div style={s.progressWrap}>
        {steps.map((label, i) => (
          <div key={label} style={{ flex: 1 }}>
            <div style={{ ...s.progressBar, background: i <= step ? "var(--teal)" : "rgba(255,255,255,0.09)" }} />
            <div style={{ ...s.progressLabel, color: i === step ? "var(--teal)" : "var(--muted)" }}>{label}</div>
          </div>
        ))}
      </div>

      <div style={s.flowBody}>
        {step === 0 && (
          <>
            <div style={s.qTitle}>Hangi konuda destek lazım?</div>
            <div style={s.catGrid}>
              {CATEGORIES.map((c) => (
                <CategoryTile key={c.key} cat={c} selected={cat === c.key} onClick={() => setCat(c.key)} />
              ))}
            </div>
            <div style={s.fieldLabel}>KONUM</div>
            <div style={s.field}>
              <label style={s.fieldSub}>Site</label>
              <input style={s.input} value={site} onChange={(e) => setSite(e.target.value)} />
            </div>
            <div style={{ display: "flex", gap: 10 }}>
              <div style={{ ...s.field, flex: 1 }}>
                <label style={s.fieldSub}>Blok</label>
                <input style={s.input} value={blok} onChange={(e) => setBlok(e.target.value)} />
              </div>
              <div style={{ ...s.field, flex: 1 }}>
                <label style={s.fieldSub}>Daire</label>
                <input style={s.input} value={daire} onChange={(e) => setDaire(e.target.value)} />
              </div>
            </div>
          </>
        )}

        {step === 1 && (
          <>
            <div style={s.qTitle}>Ustanın hazırlıklı gelmesini sağlayın</div>
            <div style={s.field}>
              <label style={s.fieldSub}>Talep başlığı</label>
              <input style={s.input} placeholder="Örn. Salon prizlerinde elektrik yok" value={title} onChange={(e) => setTitle(e.target.value)} />
            </div>
            <div style={s.field}>
              <label style={s.fieldSub}>Açıklama</label>
              <textarea style={s.textarea} rows={4} placeholder="Sorunu kısaca anlatın…" value={desc} onChange={(e) => setDesc(e.target.value)} />
            </div>
            <div style={s.fieldLabel}>ÖNCELİK</div>
            <div style={{ display: "flex", gap: 8, marginBottom: 18 }}>
              {PRIORITIES.map((p) => (
                <button key={p.key} onClick={() => setPriority(p.key)} style={{ ...s.priorityBtn, ...(priority === p.key ? s.priorityBtnActive(p.key) : {}) }}>
                  <div style={{ fontWeight: 600 }}>{p.label}</div>
                  <div style={{ fontSize: 11, opacity: 0.8 }}>{p.sub}</div>
                </button>
              ))}
            </div>
            <div style={s.fieldLabel}>TERCİH EDİLEN ZİYARET ZAMANI</div>
            <div style={{ display: "flex", flexDirection: "column", gap: 8, marginBottom: 18 }}>
              {TIME_SLOTS.map((t) => (
                <button key={t} onClick={() => setSlot(t)} style={{ ...s.slotBtn, ...(slot === t ? s.slotBtnActive : {}) }}>
                  <Clock size={15} color={slot === t ? "var(--teal)" : "var(--muted)"} />
                  {t}
                </button>
              ))}
            </div>
            <div style={s.fieldLabel}>SORUNU GÖSTERİN <span style={{ color: "var(--muted)", fontWeight: 400 }}>(isteğe bağlı)</span></div>
            <button style={s.photoBtn} onClick={() => setPhotoAdded(true)}>
              <Camera size={17} color="var(--muted)" />
              {photoAdded ? "1 fotoğraf eklendi" : "Fotoğraf / video ekle"}
            </button>
          </>
        )}

        {step === 2 && (
          <>
            <div style={s.qTitle}>Talebini gözden geçir</div>
            <div style={s.reviewCard}>
              <ReviewRow label="Hizmet" value={catObj?.label || "—"} />
              <ReviewRow label="Konum" value={`${site}, ${blok} · Daire ${daire}`} />
              <ReviewRow label="Başlık" value={title || "—"} />
              <ReviewRow label="Öncelik" value={PRIORITIES.find((p) => p.key === priority)?.label} />
              <ReviewRow label="Zaman" value={slot} last />
            </div>
            <div style={s.trustRow}>
              <ShieldCheck size={15} color="var(--teal)" />
              Kimliği doğrulanmış ustalar arasından eşleştirme yapılır.
            </div>
          </>
        )}
      </div>

      <div style={s.flowFooter}>
        <button
          style={{ ...s.submitBtn, opacity: (step === 0 && !canStep1) || (step === 1 && !canStep2) ? 0.45 : 1 }}
          disabled={(step === 0 && !canStep1) || (step === 1 && !canStep2)}
          onClick={() => (step < 2 ? setStep(step + 1) : setSent(true))}
        >
          {step < 2 ? "Devam et" : "Talebi Gönder"}
          <ChevronRight size={17} />
        </button>
      </div>
    </div>
  );
}

function statusMeta(status) {
  if (status === "active") return { label: "Yolda", color: "var(--teal)" };
  if (status === "searching") return { label: "Usta aranıyor", color: "var(--copper)" };
  return { label: "Tamamlandı", color: "var(--muted)" };
}

function Jobs({ jobs }) {
  const activeJobs = jobs.filter((j) => j.status !== "done");
  const doneJobs = jobs.filter((j) => j.status === "done");
  return (
    <div style={s.screen}>
      <TopBar title="İşlerim" />
      <div style={s.sectionPad}>
        <div style={s.sectionLabel}>Aktif talepler · {activeJobs.length}</div>
        <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
          {activeJobs.map((j) => {
            const meta = statusMeta(j.status);
            return (
              <div key={j.id} style={s.jobCard}>
                <div style={{ display: "flex", justifyContent: "space-between", alignItems: "start" }}>
                  <div style={{ fontWeight: 600 }}>{j.title}</div>
                  <span style={{ ...s.badge, color: meta.color, borderColor: meta.color }}>{meta.label}</span>
                </div>
                <div style={s.jobLoc}>{j.loc}</div>
                {j.status === "active" && (
                  <div style={s.jobFooter}>
                    <span style={{ fontSize: 12.5, color: "var(--muted)" }}>{j.master} · tahmini {j.eta}</span>
                    <button style={s.trackBtn}>Canlı Takip</button>
                  </div>
                )}
              </div>
            );
          })}
          {activeJobs.length === 0 && <div style={s.emptyNote}>Aktif talebiniz yok.</div>}
        </div>
      </div>
      {doneJobs.length > 0 && (
        <div style={s.sectionPad}>
          <div style={s.sectionLabel}>Arşiv</div>
          <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
            {doneJobs.map((j) => (
              <div key={j.id} style={{ ...s.jobCard, opacity: 0.6 }}>
                <div style={{ fontWeight: 600 }}>{j.title}</div>
                <div style={s.jobLoc}>{j.loc}</div>
              </div>
            ))}
          </div>
        </div>
      )}
      <div style={{ height: 90 }} />
    </div>
  );
}

export default function UstaCepteApp() {
  useFonts();
  const [tab, setTab] = useState("home");
  const [flowOpen, setFlowOpen] = useState(false);
  const [flowCat, setFlowCat] = useState(null);
  const [jobs, setJobs] = useState(seedJobs);
  const nextId = useRef(3);

  const startFlow = (cat) => {
    setFlowCat(cat);
    setFlowOpen(true);
  };

  const handleSubmit = (data) => {
    const id = nextId.current++;
    setJobs((prev) => [
      { id, title: data.title || data.catObj?.label || "Yeni Talep", loc: `${data.site}, ${data.blok} · Daire ${data.daire}`, status: "searching" },
      ...prev,
    ]);
    setFlowOpen(false);
    setTab("jobs");
    setTimeout(() => {
      setJobs((prev) => prev.map((j) => (j.id === id ? { ...j, status: "active", master: "Mehmet Yılmaz", eta: "12 dk" } : j)));
    }, 3200);
  };

  return (
    <div style={s.page}>
      <div style={s.phone}>
        {flowOpen ? (
          <RequestFlow initialCat={flowCat} onClose={() => setFlowOpen(false)} onSubmit={handleSubmit} />
        ) : (
          <>
            {tab === "home" && <Home onStart={startFlow} jobs={jobs} onOpenJob={() => setTab("jobs")} />}
            {tab === "jobs" && <Jobs jobs={jobs} />}
            {tab === "map" && <MapTab />}
            <div style={s.bottomNav}>
              <NavBtn icon={HomeIcon} label="Keşfet" active={tab === "home"} onClick={() => setTab("home")} />
              <NavBtn icon={MapIcon} label="Harita" active={tab === "map"} onClick={() => setTab("map")} />
              <button style={s.fab} onClick={() => startFlow(null)} aria-label="Talep Oluştur">
                <Plus size={22} color="#10161A" strokeWidth={2.5} />
              </button>
              <NavBtn icon={Briefcase} label="İşlerim" active={tab === "jobs"} onClick={() => setTab("jobs")} />
              <div style={{ width: 40 }} />
            </div>
          </>
        )}
      </div>
      <style>{`
        * { box-sizing: border-box; }
        button { font-family: inherit; cursor: pointer; }
        input, textarea { font-family: inherit; }
        ::placeholder { color: var(--muted); }
        input:focus, textarea:focus { outline: 2px solid var(--teal); outline-offset: 1px; }
        button:focus-visible { outline: 2px solid var(--teal); outline-offset: 2px; }
        @keyframes pulseRing { 0% { transform: scale(0.7); opacity: 0.7; } 100% { transform: scale(2.1); opacity: 0; } }
        @keyframes radarSpin { from { transform: rotate(0deg); } to { transform: rotate(360deg); } }
      `}</style>
    </div>
  );
}

    </button>
  );
}
