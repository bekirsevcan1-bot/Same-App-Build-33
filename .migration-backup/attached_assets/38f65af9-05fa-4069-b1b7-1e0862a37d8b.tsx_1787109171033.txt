function LiveLocationCard({ address, compact }) {
  const [coords, setCoords] = useState({ lat: 36.8874, lng: 30.7061 });
  useEffect(() => {
    const t = setInterval(() => {
      setCoords((c) => ({
        lat: c.lat + (Math.random() - 0.5) * 0.0007,
        lng: c.lng + (Math.random() - 0.5) * 0.0007,
      }));
    }, 3000);
    return () => clearInterval(t);
  }, []);
  return (
    <div style={{ ...s.liveLocCard, padding: compact ? "10px 12px" : "13px 14px" }}>
      <div style={s.liveLocPulse}>
        <span style={s.liveLocPulseRing} />
        <span style={s.liveLocPulseCore} />
      </div>
      <div style={{ flex: 1, minWidth: 0 }}>
        <div style={s.liveLocTitle}>
          <LocateFixed size={12} color="var(--teal)" /> Canlı konum paylaşılıyor
        </div>
        <div style={s.liveLocAddr}>{address}</div>
        <div style={s.liveLocCoords}>{coords.lat.toFixed(5)}° K, {coords.lng.toFixed(5)}° D</div>
      </div>
    </div>
  );
}

function MatchingScreen({ catObj, customer, onClose, onDone }) {
  const pool = useMemo(() => {
    let list = MASTERS.filter((m) => catObj && m.service.toLowerCase() === catObj.label.toLowerCase());
    if (list.length === 0) list = MASTERS.slice(0, 4);
    return [...list].sort((a, b) => parseFloat(a.distance) - parseFloat(b.distance));
  }, [catObj]);

  const [phase, setPhase] = useState("calling");
  const [visibleCount, setVisibleCount] = useState(0);
  const [matched, setMatched] = useState(null);
  const [departureTime, setDepartureTime] = useState("");

  useEffect(() => {
    let i = 0;
    const reveal = setInterval(() => {
      i += 1;
      setVisibleCount(i);
      if (i >= pool.length) clearInterval(reveal);
    }, 420);
    return () => clearInterval(reveal);
  }, [pool]);

  useEffect(() => {
    const t = setTimeout(() => {
      const chosen = pool[0] || MASTERS[0];
      const now = new Date();
      setDepartureTime(now.toLocaleTimeString("tr-TR", { hour: "2-digit", minute: "2-digit" }));
      setMatched(chosen);
      setPhase("matched");
    }, 3400);
    return () => clearTimeout(t);
  }, [pool]);

  const eta = matched ? `${Math.max(4, Math.round(parseFloat(matched.distance) * 6))} dk` : "";

  return (
    <div style={{ ...s.screen, display: "flex", flexDirection: "column" }}>
      <TopBar title="" onBack={onClose} />
      <div style={{ padding: "0 20px 12px" }}>
        <LiveLocationCard address={customer.address} compact />
      </div>

      {phase === "calling" ? (
        <div style={s.callingWrap}>
          <div style={s.radarWrap}>
            <span style={s.radarRing1} />
            <span style={s.radarRing2} />
            <div style={s.radarCore}>
              <PhoneCall size={20} color="#FFF" />
            </div>
          </div>
          <div style={{ ...s.searchTitle, marginBottom: 2 }}>
            Çevrenizdeki {catObj?.label.toLowerCase() || "hizmet"} ustaları aranıyor…
          </div>
          <div style={s.searchSub}>{customer.fullName ? `${customer.fullName} için oluşturuldu` : "Talebiniz iletiliyor"}</div>

          <div style={s.callList}>
            {pool.slice(0, visibleCount).map((m) => (
              <div key={m.name} style={s.callRow}>
                <div style={s.matchAvatar}>{m.name.split(" ").map((n) => n[0]).join("")}</div>
                <div style={{ flex: 1, textAlign: "left" }}>
                  <div style={{ fontWeight: 700, fontSize: 13.5 }}>{m.name}</div>
                  <div style={s.matchMeta}>
                    <Star size={11} color="var(--copper)" fill="var(--copper)" /> {m.rating} · {m.distance}
                  </div>
                </div>
                <span style={s.callingTag}>
                  <span style={s.callingDot} /> Çağrılıyor
                </span>
              </div>
            ))}
          </div>
        </div>
      ) : (
        <div style={s.searchWrap}>
          <div style={s.matchBadge}>
            <Check size={26} color="#FFF" strokeWidth={3} />
          </div>
          <div style={s.searchTitle}>{matched.name} talebi kabul etti!</div>
          <div style={s.searchSub}>{departureTime}'de yola çıktı</div>

          <div style={s.matchCard}>

function TrackingScreen({ job, onClose }) {
  const initialEta = parseInt(job.eta, 10) || 12;
  const [etaMin, setEtaMin] = useState(initialEta);
  const [progress, setProgress] = useState(0);

  useEffect(() => {
    const t = setInterval(() => {
      setEtaMin((m) => (m > 0 ? m - 1 : 0));
      setProgress((p) => Math.min(1, p + 1 / initialEta));
    }, 2200);
    return () => clearInterval(t);
  }, [initialEta]);

  const start = { top: 22, left: 78 };
  const end = { top: 66, left: 46 };
  const masterTop = start.top + (end.top - start.top) * progress;
  const masterLeft = start.left + (end.left - start.left) * progress;
  const arrived = etaMin <= 0;

  return (
    <div style={s.screen}>
      <TopBar title="Canlı Takip" onBack={onClose} />
      <div style={s.mapArea}>
        <div style={s.mapGrid} />
        <svg style={{ position: "absolute", inset: 0, width: "100%", height: "100%" }}>
          <line x1={`${start.left}%`} y1={`${start.top}%`} x2={`${end.left}%`} y2={`${end.top}%`} stroke="var(--navy)" strokeWidth="2" strokeDasharray="5 6" opacity="0.35" />
        </svg>
        <div style={{ ...s.pin, top: `${masterTop}%`, left: `${masterLeft}%`, background: "var(--teal)", transition: "top 2s linear, left 2s linear" }}>
          <Wrench size={14} color="#FFF" />
        </div>
        <div style={{ ...s.pin, top: `${end.top}%`, left: `${end.left}%`, background: "var(--navy)" }}>
          <HomeIcon size={14} color="#FFF" />
        </div>
        <div style={s.mapBadge}>
          <Radio size={13} color="var(--teal)" /> {arrived ? "Usta kapınızda" : `${etaMin} dk sonra varır`}
        </div>
      </div>

      <div style={s.sectionPad}>
        <div style={s.sectionLabel}>Usta bilgileri</div>
        <div style={s.jobCard}>
          <div style={{ display: "flex", alignItems: "center", gap: 12 }}>
            <div style={s.matchAvatar}>{(job.master || "U").split(" ").map((n) => n[0]).join("")}</div>
            <div style={{ flex: 1 }}>
              <div style={{ fontWeight: 700, fontSize: 14.5 }}>{job.master}</div>
              <div style={s.matchMeta}>{job.departureTime}'de yola çıktı</div>
            </div>
            <a href={`tel:${(job.masterPhone || "").replace(/\s/g, "")}`} style={s.callBtnSmall}>
              <PhoneCall size={13} /> Ara
            </a>
          </div>
        </div>
      </div>

      <div style={s.sectionPad}>
        <div style={s.sectionLabel}>Paylaşılan bilgileriniz</div>
        <div style={s.reviewCard}>
          <ReviewRow label="Ad Soyad" value={job.customerName || "—"} />
          <ReviewRow label="Telefon" value={job.customerPhone || "—"} />
          <ReviewRow label="Adres" value={job.loc} last />
        </div>
        <LiveLocationCard address={job.loc} />
      </div>
      <div style={{ height: 40 }} />
    </div>
  );
}

          </>

  );
}
